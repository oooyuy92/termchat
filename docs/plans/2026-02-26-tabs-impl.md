# Multi-Tab Chat Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add a persistent tab bar to termchat so users can run multiple independent chat sessions simultaneously, each with its own model/provider.

**Architecture:** Extract all per-tab state from `Model` into a new `TabSession` struct. `Model` holds `[]TabSession` and `activeTab int`. Stream messages carry a `TabIdx` so background tabs continue receiving output. A tab bar renders at the top of every view with mouse hit zones for click detection.

**Tech Stack:** Go, BubbleTea (bubbletea v1.3.10), lipgloss v1.1.1, bubbles/viewport, charmbracelet/x/ansi

```go
	chunks <-chan chat.StreamChunk

func (m Model) readNextChunk(tabIdx int) tea.Cmd {
	ch := m.tabs[tabIdx].streamCh
	errCh := m.tabs[tabIdx].streamErr
	return func() tea.Msg {
		select {
		case chunk, ok := <-ch:
			if !ok {
				select {
				case err := <-errCh:
					if err != nil { return streamErrMsg{TabIdx: tabIdx, Err: err} }
				default:
				}
				return streamDoneMsg{TabIdx: tabIdx}
			}
			return streamChunkMsg{TabIdx: tabIdx, Content: chunk.Content, Thinking: chunk.Thinking}
		case err := <-errCh:
			if err != nil { return streamErrMsg{TabIdx: tabIdx, Err: err} }
			return streamDoneMsg{TabIdx: tabIdx}
		}
	}
}
```

**Step 1: Define hit zone types**

```go
package ui

import (
	"fmt"
	"strings"

	"github.com/charmbracelet/lipgloss"
	xansi "github.com/charmbracelet/x/ansi"
)

type tabHitAction int

const (
	tabHitSelect   tabHitAction = iota // click tab to switch
	tabHitClose                        // click × to close
	tabHitNew                          // click + to create
	tabHitOverflow                     // click … to open overflow
)

type tabHitZone struct {
	startX int
	endX   int
	action tabHitAction
	tabIdx int // for tabHitSelect and tabHitClose
}
```

**Step 2: Implement `renderTabBar`**

```go
// renderTabBar renders the tab bar and records hit zones for mouse interaction.
// Returns the rendered string (exactly 1 terminal line wide).
func (m *Model) renderTabBar() string {
	const maxNameRunes = 14 // truncate long names
	const newBtn = " + "
	newBtnWidth := xansi.StringWidth(newBtn)

	// Pre-compute tab widths: " name × " = 1+name+3
	type tabInfo struct {
		label string // display name (may be truncated)
		width int    // total rendered width including padding
	}
	infos := make([]tabInfo, len(m.tabs))
	for i, t := range m.tabs {
		name := t.name
		runes := []rune(name)
		if len(runes) > maxNameRunes {
			name = string(runes[:maxNameRunes-1]) + "…"
		}
		label := name + " ×"
		// width = padding(1) + label + padding(1) = lipgloss padding 0,1 adds 1 each side
		w := xansi.StringWidth(label) + 2
		infos[i] = tabInfo{label: label, width: w}
	}

	// Determine how many tabs fit; reserve room for [+] and optionally […]
	overflowBtnWidth := xansi.StringWidth(" … 99 ") // max overflow label
	available := m.width - newBtnWidth
	visibleCount := 0
	used := 0
	for i, info := range infos {
		need := info.width
		if i < len(infos)-1 {
			// Need to leave room for overflow button if remaining tabs won't fit
			remaining := 0
			for _, fi := range infos[i+1:] { remaining += fi.width }
			if remaining > available-used-need {
				// This is the last visible tab; need room for overflow
				if used+need+overflowBtnWidth > available {
					break
				}
			}
		}
		if used+need > available {
			break
		}
		used += need
		visibleCount++
	}
	if visibleCount == 0 && len(infos) > 0 {
		visibleCount = 1 // always show at least one tab
	}

	hiddenCount := len(infos) - visibleCount

	// Build the bar string and record hit zones
	var sb strings.Builder
	zones := make([]tabHitZone, 0, visibleCount+2)
	x := 0

	for i := 0; i < visibleCount; i++ {
		info := infos[i]
		startX := x

		var rendered string
		if i == m.activeTab {
			rendered = m.theme.TabActiveStyle().Render(info.label)
		} else {
			rendered = m.theme.TabInactiveStyle().Render(info.label)
		}
		sb.WriteString(rendered)

		endX := x + info.width

		// Hit zone for the tab body (select)
		closeOffset := info.width - 2 // " ×" is 2 chars wide with padding
		zones = append(zones, tabHitZone{
			startX: startX,
			endX:   startX + closeOffset - 1,
			action: tabHitSelect,
			tabIdx: i,
		})
		// Hit zone for the × (close)
		zones = append(zones, tabHitZone{
			startX: startX + closeOffset,
			endX:   endX - 1,
			action: tabHitClose,
			tabIdx: i,
		})
		x = endX
	}

	// [+] button
	newRendered := lipgloss.NewStyle().
		Background(lipgloss.Color(m.theme.TabBarBg)).
		Foreground(lipgloss.Color(m.theme.TabInactiveFg)).
		Render(newBtn)
	zones = append(zones, tabHitZone{startX: x, endX: x + newBtnWidth - 1, action: tabHitNew})
	sb.WriteString(newRendered)
	x += newBtnWidth

	// […] overflow button (if any hidden)
	if hiddenCount > 0 {
		overflowLabel := fmt.Sprintf(" … %d ", hiddenCount)
		overflowW := xansi.StringWidth(overflowLabel)
		overflowRendered := lipgloss.NewStyle().
			Background(lipgloss.Color(m.theme.TabBarBg)).
			Foreground(lipgloss.Color(m.theme.TabInactiveFg)).
			Render(overflowLabel)
		zones = append(zones, tabHitZone{startX: x, endX: x + overflowW - 1, action: tabHitOverflow})
		sb.WriteString(overflowRendered)
		x += overflowW
	}

	m.tabBarZones = zones

	// Pad to full width with tab bar background
	bar := lipgloss.NewStyle().
		Background(lipgloss.Color(m.theme.TabBarBg)).
		Width(m.width).
		Render(sb.String())

	return bar
}

// hitTestTabBar returns the hit zone at column x on the tab bar row (row 0).
// Returns nil if no zone matches.
func (m *Model) hitTestTabBar(x int) *tabHitZone {
	for i := range m.tabBarZones {
		z := &m.tabBarZones[i]
		if x >= z.startX && x <= z.endX {
			return z
		}
	}
	return nil
}
```

```go
func (m *Model) closeTab(idx int) {
	if len(m.tabs) <= 1 {
		m.statusMsg = "Cannot close last tab"
		return
	}
	// Cancel in-flight stream
	if m.tabs[idx].streamCtrl != nil {
		m.tabs[idx].streamCtrl.cancel()
	}
	m.tabs = append(m.tabs[:idx], m.tabs[idx+1:]...)
	if m.activeTab >= len(m.tabs) {

	} else if m.activeTab > idx {

}
```

**Step 2: Create `taboverflow.go`**

```go
package ui

import (
	"fmt"
	"strings"

	tea "github.com/charmbracelet/bubbletea"
	"github.com/charmbracelet/lipgloss"
)

func (m Model) updateTabOverflow(msg tea.KeyMsg) (Model, tea.Cmd) {
	// visibleCount = number of tabs shown in tab bar (computed during last render)
	visibleCount := 0
	for _, z := range m.tabBarZones {
		if z.action == tabHitSelect && z.tabIdx+1 > visibleCount {
			visibleCount = z.tabIdx + 1
		}
	}
	hiddenTabs := m.tabs[visibleCount:]
	n := len(hiddenTabs)

	switch msg.String() {
	case "esc", "alt+0":
		m.mode = modeChat
	case "up", "k":
		if m.tabOverflowOffset > 0 { m.tabOverflowOffset-- }
	case "down", "j":
		if m.tabOverflowOffset < n-1 { m.tabOverflowOffset++ }
	case "enter":
		m.switchTab(visibleCount + m.tabOverflowOffset)
		m.tabOverflowOffset = 0
		m.mode = modeChat
	}
	return m, nil
}

func (m Model) viewTabOverflow() string {
	visibleCount := 0
	for _, z := range m.tabBarZones {
		if z.action == tabHitSelect && z.tabIdx+1 > visibleCount {
			visibleCount = z.tabIdx + 1
		}
	}
	hiddenTabs := m.tabs[visibleCount:]

	var lines []string
	for i, t := range hiddenTabs {
		cursor := "  "
		if i == m.tabOverflowOffset {
			cursor = m.theme.ConfigCursorStyle().Render("> ")
		}
		idx := visibleCount + i
		lines = append(lines, fmt.Sprintf("%s[%d] %s", cursor, idx+1, t.name))
	}
	dropdown := strings.Join(lines, "\n")

	tabBar := (&m).renderTabBar()
	statusBar := m.renderStatusBar()
	inputArea := m.tabs[m.activeTab].textarea.View()
	vpHeight := m.height - lipgloss.Height(tabBar) - lipgloss.Height(statusBar) -
		lipgloss.Height(inputArea) - lipgloss.Height(dropdown)
	if vpHeight < 1 { vpHeight = 1 }
	m.tabs[m.activeTab].viewport.Height = vpHeight

	parts := []string{tabBar, m.tabs[m.activeTab].viewport.View(), dropdown, inputArea, statusBar}
	return lipgloss.JoinVertical(lipgloss.Left, parts...)
}
```
